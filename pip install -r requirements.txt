# =============================================================================
# config.py  ·  환경 변수 로드 & API 키 검증
# =============================================================================
# 이 파일이 하는 일:
#   1) 프로젝트 루트의 .env 파일을 읽어 환경 변수를 메모리에 올립니다.
#   2) 필수 API 키 2개(Gemini, Kakao)가 존재하는지 검증합니다.
#   3) 키가 없으면 무엇이 문제인지 알려주고 프로그램을 안전하게 종료합니다.
# =============================================================================

import os
import sys
from dotenv import load_dotenv  # pip install python-dotenv

# -----------------------------------------------------------------------------
# 1. .env 파일 로드
# -----------------------------------------------------------------------------
# load_dotenv()는 현재 디렉터리(또는 상위)에서 .env 파일을 찾아
# 그 안의 KEY=VALUE 쌍을 os.environ 에 등록해 줍니다.
load_dotenv()

# -----------------------------------------------------------------------------
# 2. 환경 변수에서 API 키 읽기
# -----------------------------------------------------------------------------
# os.getenv("키이름")은 해당 환경 변수가 없으면 None 을 반환합니다.
GEMINI_API_KEY    = os.getenv("GEMINI_API_KEY")
KAKAO_REST_API_KEY = os.getenv("KAKAO_REST_API_KEY")

# -----------------------------------------------------------------------------
# 3. 필수 키 검증
# -----------------------------------------------------------------------------
def validate_keys() -> None:
    """
    API 키가 모두 설정되어 있는지 확인합니다.
    하나라도 빠져 있으면 어떤 키가 없는지 출력하고 프로그램을 종료합니다.
    """

    # 검사할 키 목록: (변수값, 표시할 키 이름) 형태로 관리합니다.
    required_keys = [
        (GEMINI_API_KEY,     "GEMINI_API_KEY"),
        (KAKAO_REST_API_KEY, "KAKAO_REST_API_KEY"),
    ]

    # 값이 None 이거나 빈 문자열("")인 키를 모아 둡니다.
    missing = [name for value, name in required_keys if not value]

    if missing:
        # ── 오류 메시지 출력 ──────────────────────────────────────────────
        print("\n❌  API 키가 설정되지 않았습니다.")
        print("─" * 45)
        for key_name in missing:
            print(f"   • {key_name}  ← 이 키가 없습니다")
        print("─" * 45)

        # ── 해결 방법 안내 ────────────────────────────────────────────────
        print("\n📋  해결 방법:")
        print("   1) 프로젝트 루트에 '.env' 파일을 만드세요.")
        print("      (참고: .env.example 파일을 복사하면 편합니다)")
        print()
        print("   2) 아래 형식으로 키를 입력하세요:")
        for key_name in missing:
            print(f"      {key_name}=여기에_실제_키_입력")
        print()
        print("   3) 키 발급 방법:")
        print("      • Gemini  → https://aistudio.google.com/app/apikey")
        print("      • Kakao   → https://developers.kakao.com  (내 애플리케이션)")
        print()

        # sys.exit(1) : 비정상 종료를 의미하는 종료 코드 1로 프로그램을 끝냅니다.
        sys.exit(1)


# -----------------------------------------------------------------------------
# 4. 이 파일을 직접 실행했을 때 동작 (테스트용)
# -----------------------------------------------------------------------------
# `python config.py` 로 실행하면 키 검증 결과를 바로 확인할 수 있습니다.
if __name__ == "__main__":
    validate_keys()
    print("✅  모든 API 키가 정상적으로 로드되었습니다!")
    print(f"   GEMINI_API_KEY     : {GEMINI_API_KEY[:6]}{'*' * 10}")   # 앞 6자만 표시
    print(f"   KAKAO_REST_API_KEY : {KAKAO_REST_API_KEY[:6]}{'*' * 10}")
